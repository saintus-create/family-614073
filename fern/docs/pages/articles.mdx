function ArticleSearch() {
  const [query, setQuery] = useState("");
  const [selected, setSelected] = useState(null);

  const filtered = articles.filter(
    (a) =>
      a.title.toLowerCase().includes(query.toLowerCase()) ||
      a.journal.toLowerCase().includes(query.toLowerCase()) ||
      a.medication.toLowerCase().includes(query.toLowerCase()) ||
      a.doi.toLowerCase().includes(query.toLowerCase())
  );

  return (
    <div>
      <div className="search-bar">
        <input
          type="text"
          placeholder="Search articles by title, journal, medication, or DOI..."
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          className="search-input"
        />
      </div>
      <p className="result-count">
        {filtered.length} of {articles.length} articles
      </p>
      <div className="article-list">
        {filtered.map((article, idx) => (
          <div
            key={idx}
            className="article-card"
            onClick={() => setSelected(selected === idx ? null : idx)}
          >
            <h3>{article.title}</h3>
            <p className="article-meta">
              <strong>{article.journal}</strong> ({article.year}) &middot; DOI: {article.doi}
            </p>
            <p className="article-meta">
              Medication: {article.medication} &middot; Usage: {article.usageType}
            </p>
            {selected === idx && (
              <div className="article-detail">
                <p><strong>Key Findings:</strong> {article.keyFindings}</p>
                <a href={article.sourceUrl} target="_blank" rel="noopener noreferrer">
                  View Source &rarr;
                </a>
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
